cmake_minimum_required(VERSION 3.20)
project(skottie-render CXX)
set(CMAKE_CXX_STANDARD 20)
set(SKIA_DIR "$ENV{HOME}/Projects/VideoEd/third_party/prebuilt/skia/linux-x64" CACHE PATH "Directory containing SkiaConfig.cmake")
include(${SKIA_DIR}/SkiaConfig.cmake)
add_executable(skottie-render main.cpp)
target_compile_definitions(skottie-render PRIVATE ${SKIA_DEFINES})
target_include_directories(skottie-render PRIVATE ${SKIA_DIR}/include)
target_link_libraries(skottie-render PRIVATE ${SKIA_LIBRARIES} ${SKIA_SYSTEM_LIBRARIES})
